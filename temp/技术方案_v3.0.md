## 1. 文档概述

### 1.1 文档目的

本文档为智能云备份归档系统的技术方案设计文档，基于PRD v3.0需求文档编写，旨在为开发团队提供详细的技术实现指导。

### 1.2 适用范围

本文档适用于以下场景：
- 系统架构设计评审
- 模块接口定义
- 数据库表结构实现
- 百度网盘上传功能开发
- 任务状态机与调度器开发

### 1.3 术语定义

| 术语 | 定义 |
|------|------|
| Task | 备份任务，对应数据库中的一条BackupRecord |
| Block | 分片，百度网盘上传的最小数据单元（默认4MB） |
| Session | 会话ID，用于崩溃恢复和任务追踪 |
| BlockList | 分片MD5列表，用于百度秒传预检查 |

```python
# 状态允许的操作映射
```

### 4.1 任务状态机

#### 4.1.1 状态机类设计

```python
from abc import ABC, abstractmethod
from typing import Dict, Set, Optional, Callable
from dataclasses import dataclass
from datetime import datetime
import asyncio

@dataclass
class StateTransition:
    """状态转换记录"""
    from_status: TaskStatus
    to_status: TaskStatus
    operation: TaskOperation
    timestamp: datetime
    reason: Optional[str] = None


class TaskStateMachine:
    """任务状态机"""

    def __init__(self, task_id: int):
        self.task_id = task_id
        self._current_status: TaskStatus = TaskStatus.NEW
        self._transition_history: list[StateTransition] = []
        self._lock = asyncio.Lock()
        self._callbacks: Dict[TaskStatus, list[Callable]] = {}

    @property
    def current_status(self) -> TaskStatus:
        return self._current_status

    async def can_perform(self, operation: TaskOperation) -> bool:
        """检查是否允许执行某个操作"""
        async with self._lock:
            allowed = STATUS_OPERATIONS.get(self._current_status, set())
            return operation in allowed

    async def transition(
        self,
        operation: TaskOperation,
        to_status: TaskStatus,
        reason: Optional[str] = None
    ) -> bool:
        """
        执行状态转换

        Args:
            operation: 执行的操作
            to_status: 目标状态
            reason: 转换原因

        Returns:
            是否转换成功
        """
        async with self._lock:
            # 验证操作是否允许
            if operation not in STATUS_OPERATIONS.get(self._current_status, set()):
                return False

            # 记录转换
            transition = StateTransition(
                from_status=self._current_status,
                to_status=to_status,
                operation=operation,
                timestamp=datetime.now(),
                reason=reason
            )
            self._transition_history.append(transition)

            # 执行回调
            old_status = self._current_status
            self._current_status = to_status

            # 触发状态变更回调
            if to_status in self._callbacks:
                for callback in self._callbacks[to_status]:
                    await callback(self.task_id, old_status, to_status)

            return True

    def on_status_change(
        self,
        status: TaskStatus,
        callback: Callable[[int, TaskStatus, TaskStatus], None]
    ):
        """注册状态变更回调"""
        if status not in self._callbacks:
            self._callbacks[status] = []
        self._callbacks[status].append(callback)

    def get_transition_history(self) -> list[StateTransition]:
        """获取转换历史"""
        return self._transition_history.copy()
```

#### 4.1.2 状态机管理器

```python
class StateMachineManager:
    """状态机管理器 - 单例模式"""

    def __init__(self):
        self._machines: Dict[int, TaskStateMachine] = {}
        self._lock = asyncio.Lock()

    async def get_machine(self, task_id: int) -> TaskStateMachine:
        """获取任务状态机"""
        async with self._lock:
            if task_id not in self._machines:
                self._machines[task_id] = TaskStateMachine(task_id)
            return self._machines[task_id]

    async def remove_machine(self, task_id: int):
        """移除状态机"""
        async with self._lock:
            if task_id in self._machines:
                del self._machines[task_id]

    async def get_status(self, task_id: int) -> Optional[TaskStatus]:
        """获取任务当前状态"""
        machine = await self.get_machine(task_id)
        return machine.current_status

    async def can_perform_operation(
        self,
        task_id: int,
        operation: TaskOperation
    ) -> bool:
        """检查任务是否可以执行操作"""
        machine = await self.get_machine(task_id)
        return await machine.can_perform(operation)
```

### 4.2 任务调度器

#### 4.2.1 调度器核心类

```python
from dataclasses import dataclass, field
from typing import Deque, Set, Dict, Optional
from collections import deque

@dataclass
class SchedulerConfig:
    """调度器配置"""
    max_concurrent_tasks: int = 3
    max_disk_usage_bytes: int = 10 * 1024 ** 3  # 10GB
    default_block_size: int = 4 * 1024 ** 2  # 4MB
    max_retry_count: int = 3
    pause_check_interval: float = 1.0  # 暂停检查间隔


class TaskScheduler:
    """任务调度器"""

    def __init__(self, config: SchedulerConfig):
        self.config = config
        self._active_tasks: Set[int] = set()  # 正在执行的任务
        self._queued_tasks: Deque[int] = deque()  # 排队任务
        self._paused_tasks: Set[int] = set()  # 暂停的任务
        self._task_semaphore: asyncio.Semaphore = asyncio.Semaphore(
            config.max_concurrent_tasks
        )
        self._state_machine_manager = StateMachineManager()
        self._running = False
        self._cancel_events: Dict[int, asyncio.Event] = {}  # 取消事件
        self._lock = asyncio.Lock()

    async def start(self):
        """启动调度器"""
        self._running = True
        asyncio.create_task(self._process_queue_loop())

    async def stop(self):
        """停止调度器"""
        self._running = False

    async def submit(self, task_id: int) -> bool:
        """
        提交任务到队列

        Args:
            task_id: 任务ID

        Returns:
            是否提交成功
        """
        async with self._lock:
            # 检查状态
            status = await self._state_machine_manager.get_status(task_id)
            if status != TaskStatus.NEW:
                return False

            # 添加到队列
            self._queued_tasks.append(task_id)
            self._cancel_events[task_id] = asyncio.Event()

            # 更新状态
            await self._state_machine_manager.get_machine(task_id).transition(
                TaskOperation.SUBMIT,
                TaskStatus.QUEUED,
                "任务已提交到队列"
            )

            return True

    async def pause(self, task_id: int, reason: str = "用户暂停") -> bool:
        """
        暂停任务

        Args:
            task_id: 任务ID
            reason: 暂停原因
        """
        async with self._lock:
            # 验证操作
            can_pause = await self._state_machine_manager.can_perform_operation(
                task_id, TaskOperation.PAUSE
            )
            if not can_pause:
                return False

            # 如果任务正在执行，发送取消信号
            if task_id in self._active_tasks:
                self._cancel_events[task_id].set()
                self._active_tasks.discard(task_id)

            # 从队列中移除
            self._queued_tasks = deque(
                [t for t in self._queued_tasks if t != task_id]
            )

            # 标记为暂停
            self._paused_tasks.add(task_id)

            machine = await self._state_machine_manager.get_machine(task_id)
            await machine.transition(
                TaskOperation.PAUSE,
                TaskStatus.PAUSED,
                reason
            )

            return True

    async def resume(self, task_id: int) -> bool:
        """
        恢复任务

        Args:
            task_id: 任务ID
        """
        async with self._lock:
            if task_id not in self._paused_tasks:
                return False

            self._paused_tasks.discard(task_id)
            self._queued_tasks.append(task_id)
            self._cancel_events[task_id] = asyncio.Event()

            machine = await self._state_machine_manager.get_machine(task_id)
            await machine.transition(
                TaskOperation.RESUME,
                TaskStatus.QUEUED,
                "任务已恢复"
            )

            return True

    async def cancel(self, task_id: int, reason: str = "用户取消") -> bool:
        """
        取消任务

        Args:
            task_id: 任务ID
            reason: 取消原因
        """
        async with self._lock:
            can_cancel = await self._state_machine_manager.can_perform_operation(
                task_id, TaskOperation.CANCEL
            )
            if not can_cancel:
                return False

            # 发送取消信号
            if task_id in self._cancel_events:
                self._cancel_events[task_id].set()

            # 从各集合中移除
            self._active_tasks.discard(task_id)
            self._queued_tasks = deque(
                [t for t in self._queued_tasks if t != task_id]
            )
            self._paused_tasks.discard(task_id)

            # 更新状态
            machine = await self._state_machine_manager.get_machine(task_id)
            await machine.transition(
                TaskOperation.CANCEL,
                TaskStatus.CANCELLED,
                reason
            )

            return True

    async def retry(self, task_id: int) -> bool:
        """
        重试任务

        Args:
            task_id: 任务ID
        """
        async with self._lock:
            can_retry = await self._state_machine_manager.can_perform_operation(
                task_id, TaskOperation.RETRY
            )
            if not can_retry:
                return False

            self._queued_tasks.append(task_id)
            self._cancel_events[task_id] = asyncio.Event()

            machine = await self._state_machine_manager.get_machine(task_id)
            await machine.transition(
                TaskOperation.RETRY,
                TaskStatus.QUEUED,
                f"用户请求重试 (第{await self._get_retry_count(task_id) + 1}次)"
            )

            return True

    async def _process_queue_loop(self):
        """处理队列循环"""
        while self._running:
            if self._queued_tasks and len(self._active_tasks) < self.config.max_concurrent_tasks:
                task_id = self._queued_tasks.popleft()

                if task_id not in self._active_tasks and task_id not in self._paused_tasks:
                    self._active_tasks.add(task_id)
                    asyncio.create_task(self._execute_task(task_id))

            await asyncio.sleep(self.config.pause_check_interval)

    async def _execute_task(self, task_id: int):
        """执行任务"""
        try:
            # 更新状态为PENDING
            machine = await self._state_machine_manager.get_machine(task_id)
            await machine.transition(
                TaskOperation.SUBMIT,
                TaskStatus.PENDING,
                "开始执行任务"
            )

            # 获取取消事件
            cancel_event = self._cancel_events.get(task_id)
            if not cancel_event:
                cancel_event = asyncio.Event()
                self._cancel_events[task_id] = cancel_event

            # 创建任务执行器并执行
            executor = TaskExecutor(task_id, cancel_event)
            await executor.execute()

        except Exception as e:
            # 任务执行失败
            await self._handle_task_failure(task_id, str(e))
        finally:
            async with self._lock:
                self._active_tasks.discard(task_id)

    async def _handle_task_failure(self, task_id: int, error: str):
        """处理任务失败"""
        machine = await self._state_machine_manager.get_machine(task_id)
        retry_count = await self._get_retry_count(task_id)
        max_retries = await self._get_max_retries(task_id)

        if retry_count < max_retries:
            # 可以重试，标记为FAILED状态等待用户操作
            await machine.transition(
                TaskOperation.CANCEL,
                TaskStatus.FAILED,
                f"任务失败: {error}"
            )
        else:
            # 超过最大重试次数
            await machine.transition(
                TaskOperation.CANCEL,
                TaskStatus.FAILED,
                f"任务失败，已达最大重试次数: {error}"
            )

    async def _get_retry_count(self, task_id: int) -> int:
```

```python
import hashlib
import os
from typing import List, Generator, Optional
from dataclasses import dataclass

@dataclass
class BlockInfo:
    """分片信息"""
    block_number: int
    md5: str
    offset: int
    size: int


class BlockMD5Calculator:
    """分片MD5计算器"""

    def __init__(self, block_size: int = 4 * 1024 * 1024):
        self.block_size = block_size

    def calculate_file_block_md5_list(
        self,
        file_path: str
    ) -> List[str]:
        """
        计算文件的分片MD5列表

        Args:
            file_path: 文件路径

        Returns:
            MD5十六进制字符串列表
        """
        md5_list = []

        file_size = os.path.getsize(file_path)
        total_blocks = (file_size + self.block_size - 1) // self.block_size

        for block_num in range(total_blocks):
            offset = block_num * self.block_size
            size = min(self.block_size, file_size - offset)

            with open(file_path, 'rb') as f:
                f.seek(offset)
                chunk = f.read(size)

            md5_hash = hashlib.md5(chunk).hexdigest()
            md5_list.append(md5_hash)

        return md5_list

    def calculate_file_block_md5_generator(
        self,
        file_path: str
    ) -> Generator[BlockInfo, None, None]:
        """
        生成器方式计算分片MD5（内存友好）

        Args:
            file_path: 文件路径

        Yields:
            BlockInfo: 分片信息
        """
        file_size = os.path.getsize(file_path)
        total_blocks = (file_size + self.block_size - 1) // self.block_size

        with open(file_path, 'rb') as f:
            for block_num in range(total_blocks):
                offset = block_num * self.block_size
                size = min(self.block_size, file_size - offset)

                chunk = f.read(size)
                md5_hash = hashlib.md5(chunk).hexdigest()

                yield BlockInfo(
                    block_number=block_num + 1,  # 百度API从1开始
                    md5=md5_hash,
                    offset=offset,
                    size=size
                )

    @staticmethod
    def encode_block_list(md5_list: List[str]) -> str:
        """
        将MD5列表编码为逗号分隔的字符串

        Args:
            md5_list: MD5字符串列表

        Returns:
            逗号分隔的字符串
        """
        return ','.join(md5_list)

    @staticmethod
    def decode_block_list(encoded: str) -> List[str]:
        """
        解码MD5列表字符串

        Args:
            encoded: 逗号分隔的字符串

        Returns:
            MD5字符串列表
        """
        if not encoded:
            return []
        return encoded.split(',')
```

```python
class BaiduPanUploader:
    """百度网盘上传服务"""

    def __init__(
        self,
        client: BaiduPanClient,
        block_size: int = 4 * 1024 ** 2
    ):
        self.client = client
        self.block_size = block_size
        self.block_calculator = BlockMD5Calculator(block_size)

    async def upload_file(
        self,
        local_path: str,
        cloud_path: str,
        upload_id: Optional[str] = None,
        progress_callback: Optional[callable] = None
    ) -> UploadResult:
        """
        上传文件

        Args:
            local_path: 本地文件路径
            cloud_path: 云端目标路径
            upload_id: 断点续传ID(可选)
            progress_callback: 进度回调函数

        Returns:
            UploadResult: 上传结果
        """
        file_size = os.path.getsize(local_path)

        try:
            # Step 1: 计算分片MD5列表
            block_md5_list = self.block_calculator.calculate_file_block_md5_list(
                local_path
            )

            # Step 2: 预创建上传
            precreate_result = await self.client.precreate(
                cloud_path=cloud_path,
                file_size=file_size,
                block_list=block_md5_list,
                upload_id=upload_id
            )

            new_upload_id = precreate_result["upload_id"]
            block_seq = precreate_result.get("block_seq", [])

            # 如果block_seq为空，说明所有分片都已存在(秒传)
            if not block_seq:
                return UploadResult(
                    success=True,
                    cloud_path=cloud_path,
                    uploaded_bytes=file_size,
                    total_bytes=file_size,
                    file_id=precreate_result.get("fs_id")
                )

            # Step 3: 上传缺失的分片
            uploaded_bytes = 0
            for block_num in block_seq:
                # 读取分片数据
                offset = (block_num - 1) * self.block_size
                size = min(self.block_size, file_size - offset)

                with open(local_path, 'rb') as f:
                    f.seek(offset)
                    chunk = f.read(size)

                # 上传分片
                await self.client.upload_part(
                    upload_id=new_upload_id,
                    part_number=block_num,
                    data=chunk
                )

                uploaded_bytes += size

                # 回调进度
                if progress_callback:
                    progress_callback(uploaded_bytes, file_size)

            # Step 4: 完成上传
            finish_result = await self.client.finish_upload(
                cloud_path=cloud_path,
                upload_id=new_upload_id,
                block_list=block_md5_list
            )

            return UploadResult(
                success=True,
                file_id=finish_result.get("fs_id"),
                cloud_path=cloud_path,
                uploaded_bytes=uploaded_bytes,
                total_bytes=file_size
            )

        except Exception as e:
            return UploadResult(
                success=False,
                error_message=str(e),
                uploaded_bytes=0,
                total_bytes=file_size
            )
```
